# Runtime dependencies for EmAtSy v3.
#
# This file used to list the Django stack - Django, channels, daphne, Twisted,
# autobahn, djangorestframework - none of which v3 uses. That was carried over
# from the `fast_api/` implementation removed on 2026-08-25, and it broke a
# fresh install outright: autobahn==25.10.2 requires Python >=3.11, while the
# compiled core is built for CPython 3.10.
#
# Derived from what the code actually imports. Install onnxruntime-gpu
# SEPARATELY with --no-deps (see below): pip resolves the CPU package as a
# dependency of other things and it silently shadows the GPU build, costing
# roughly 10x inference speed.

# --- web layer -------------------------------------------------------------
fastapi==0.141.1
starlette==0.50.0
uvicorn==0.52.4
# NOT optional despite pip treating it as such: plain `uvicorn` has no
# WebSocket transport, so `ws="auto"` resolves to none and every /ws/...
# handshake is served as an ordinary GET - which the auth middleware answers
# with a 303 to /login. The live camera panels then show "Mavjud emas" forever
# with nothing logged. Pure Python, no dependencies of its own, so it is safe
# to install with --no-deps on the GPU host.
websockets==16.1.1
jinja2==3.1.5
MarkupSafe==3.0.2
python-multipart==0.0.32       # multipart form uploads on the enrolment page
